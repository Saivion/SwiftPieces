/// A pile of photos that fans out into a grid and gathers back.
///
/// It takes the width it is offered and grows to the height the grid needs, so put it in a `ScrollView` or a
/// `VStack`. The first item lies on top of the pile.
///
/// - Parameters:
///   - items: The photos, first on top.
///   - isExpanded: Whether they are laid out as a grid. A tap on the pile sets it.
///   - columns: How many columns the grid has.
///   - countLabel: The words under the pile, from the count. Pass `nil` to hide them.
///   - style: Sizes and colors.
///   - onSelect: Called with a photo tapped in the grid. Without it, a tap in the grid gathers the pile.
///   - content: One photo. It is clipped to the style's corner radius and sized to fill.
public struct PhotoStack<Item: Identifiable, Content: View>: View {
    public typealias Style = PhotoStackStyle

    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    @Binding private var isExpanded: Bool
    @State private var width: CGFloat = 0
    @State private var toggles = 0

    private let items: [Item]
    private let columns: Int
    private let countLabel: ((Int) -> String)?
    private let style: Style
    private let onSelect: ((Item) -> Void)?
    private let content: (Item) -> Content

    /// The pile: each photo's tilt and nudge, by its place from the top. Photos past the fourth hide under it.
    private static var tilts: [Double] { [0, -4, 7, -9] }
    private static var nudges: [CGSize] { [.zero, CGSize(width: 2, height: 6), CGSize(width: 6, height: -4), CGSize(width: -7, height: 5)] }

    public init(
        _ items: [Item],
        isExpanded: Binding<Bool>,
        columns: Int = 2,
        countLabel: ((Int) -> String)? = { "\($0) photos" },
        style: Style = .standard,
        onSelect: ((Item) -> Void)? = nil,
        @ViewBuilder content: @escaping (Item) -> Content
    ) {
        self.items = items
        self._isExpanded = isExpanded
        self.columns = max(1, columns)
        self.countLabel = countLabel
        self.style = style
        self.onSelect = onSelect
        self.content = content
    }

    public var body: some View {
        let motion = PieceMotion(reduceMotion: reduceMotion)
        let rows = (items.count + columns - 1) / columns
        let cell = width > 0 ? (width - style.spacing * CGFloat(columns - 1)) / CGFloat(columns) : style.stackSize
        let gridHeight = CGFloat(rows) * cell + CGFloat(max(rows - 1, 0)) * style.spacing
        let pileHeight = style.stackSize + 24
        let height = isExpanded ? gridHeight : pileHeight

        ZStack(alignment: .top) {
            ForEach(Array(items.enumerated()), id: \.element.id) { index, item in
                let place = placement(index: index, cell: cell)
                content(item)
                    .frame(width: place.size, height: place.size)
                    .clipShape(.rect(cornerRadius: style.cornerRadius, style: .continuous))
                    .shadow(color: style.shadow.opacity(isExpanded ? 0.5 : 1), radius: isExpanded ? 4 : 10, y: isExpanded ? 2 : 5)
                    .rotationEffect(.degrees(place.tilt))
                    .offset(place.offset)
                    .opacity(isExpanded || index < 4 ? 1 : 0)
                    .zIndex(Double(items.count - index))
                    .animation(isExpanded ? motion.follow(rank: index) : motion.follow(PieceMotion.responsive, rank: items.count - 1 - index), value: isExpanded)
                    .contentShape(.rect)
                    .onTapGesture { tap(item) }
                    .accessibilityHidden(!isExpanded)
            }
            if let countLabel {
                Text(countLabel(items.count))
                    .font(.footnote.weight(.semibold))
                    .foregroundStyle(style.label)
                    .offset(y: style.stackSize + 14)
                    .opacity(isExpanded ? 0 : 1)
                    .animation(motion.reveal, value: isExpanded)
                    .accessibilityHidden(true)
            }
        }
        .frame(maxWidth: .infinity)
        .frame(height: height, alignment: .top)
        .animation(isExpanded ? motion.reveal : motion.dismiss, value: isExpanded)
        .onGeometryChange(for: CGFloat.self) { $0.size.width } action: { width = $0 }
        .sensoryFeedback(.impact(weight: .light), trigger: toggles)
        .accessibilityElement(children: isExpanded ? .contain : .ignore)
        .accessibilityLabel(isExpanded ? "" : (countLabel?(items.count) ?? "\(items.count) photos"))
        .accessibilityAddTraits(isExpanded ? [] : .isButton)
        .accessibilityHint(isExpanded ? "" : "Shows them as a grid")
        .accessibilityAction(named: isExpanded ? "Stack" : "Show all") { toggle() }
    }

    /// Where photo `index` sits: in the pile, centred, tilted and nudged; in the grid, in its cell.
    private func placement(index: Int, cell: CGFloat) -> (size: CGFloat, tilt: Double, offset: CGSize) {
        if isExpanded {
            let column = index % columns, row = index / columns
            let x = (CGFloat(column) - CGFloat(columns - 1) / 2) * (cell + style.spacing)
            let y = CGFloat(row) * (cell + style.spacing)
            return (cell, 0, CGSize(width: x, height: y))
        }
        let rank = min(index, 3)
        return (style.stackSize, Self.tilts[rank], Self.nudges[rank])
    }

    private func tap(_ item: Item) {
        if isExpanded, let onSelect { onSelect(item) } else { toggle() }
    }

    private func toggle() {
        toggles += 1
        isExpanded.toggle()
    }
}

/// A flat print in the house palette, standing in for a photo in the example. Swap in your own images.
private struct PhotoStackPrint: View {
    /// Which of the eight prints: 0 Sunrise, 1 Lake, 2 Pines, 3 Waves, 4 Dunes, 5 Night, 6 Bloom, 7 Kite.
    var kind: Int

    private static let blue = Color(red: 0.18, green: 0.545, blue: 1)
    private static let yellow = Color(red: 1, green: 0.761, blue: 0.102)
    private static let teal = Color(red: 0, green: 0.769, blue: 0.655)
    private static let pink = Color(red: 1, green: 0, blue: 0.361)
    private static let orange = Color(red: 1, green: 0.541, blue: 0.122)
    private static let red = Color(red: 1, green: 0, blue: 0)
    private static let ink = Color(red: 0.078, green: 0.078, blue: 0.078)
    private static let paper = Color(red: 0.957, green: 0.953, blue: 0.937)

    var body: some View {
        Canvas { context, size in
            let w = size.width, h = size.height, m = min(w, h)
            func disc(_ x: CGFloat, _ y: CGFloat, _ r: CGFloat, _ color: Color) {
                context.fill(Path(ellipseIn: CGRect(x: x - r, y: y - r, width: r * 2, height: r * 2)), with: .color(color))
            }
            func band(_ y: CGFloat, _ height: CGFloat, _ color: Color) {
                context.fill(Path(CGRect(x: 0, y: y, width: w, height: height)), with: .color(color))
            }
            switch kind % 8 {
            case 0:
                band(0, h, Self.orange)
                disc(w * 0.5, h * 0.64, m * 0.34, Self.yellow)
                for (y, t) in [(0.66, 0.035), (0.74, 0.05), (0.83, 0.07), (0.93, 0.07)] as [(CGFloat, CGFloat)] { band(h * y, h * t, Self.pink) }
            case 1:
                band(0, h, Self.blue)
                disc(w * 0.7, h * 0.32, m * 0.13, Self.yellow)
                band(h * 0.6, h * 0.4, Self.teal)
                for (a, b, l) in [(0.2, 0.72, 0.22), (0.55, 0.8, 0.28), (0.3, 0.9, 0.18)] as [(CGFloat, CGFloat, CGFloat)] {
                    var ripple = Path()
                    ripple.move(to: CGPoint(x: w * a, y: h * b))
                    ripple.addLine(to: CGPoint(x: w * (a + l), y: h * b))
                    context.stroke(ripple, with: .color(Self.paper), style: StrokeStyle(lineWidth: max(1, m * 0.025), lineCap: .round))
                }
            case 2:
                band(0, h, Self.teal)
                disc(w * 0.3, h * 0.32, m * 0.17, Self.pink)
                let count = max(4, Int((w / m * 5).rounded()))
                for i in 0..<count {
                    let x = (CGFloat(i) + 0.5) / CGFloat(count) * w, half = w / CGFloat(count) * 0.62
                    let height = h * (0.34 + CGFloat((i * 37) % 5) / 5 * 0.2)
                    var pine = Path()
                    pine.move(to: CGPoint(x: x, y: h - height))
                    pine.addLine(to: CGPoint(x: x + half, y: h))
                    pine.addLine(to: CGPoint(x: x - half, y: h))
                    pine.closeSubpath()
                    context.fill(pine, with: .color(Self.ink))
                }
            case 3:
                band(0, h, Self.pink)
                disc(w * 0.72, h * 0.3, m * 0.16, Self.yellow)
                for (j, color) in [Self.blue, Self.teal].enumerated() {
                    let r = m * 0.1, y = h * (0.62 + CGFloat(j) * 0.16)
                    var waves = Path()
                    waves.move(to: CGPoint(x: -r * 3, y: h))
                    waves.addLine(to: CGPoint(x: -r * 3, y: y + r))
                    var x = -r * 2 + CGFloat(j) * r
                    while x < w + r * 3 {
                        waves.addArc(center: CGPoint(x: x, y: y + r), radius: r, startAngle: .degrees(180), endAngle: .degrees(0), clockwise: false)
                        x += r * 2
                    }
                    waves.addLine(to: CGPoint(x: w + r * 3, y: h))
                    waves.closeSubpath()
                    context.fill(waves, with: .color(color))
                }
            case 4:
                band(0, h, Self.yellow)
                disc(w * 0.66, h * 0.3, m * 0.1, Self.red)
                disc(w * 0.18, h + m * 0.25, m * 0.62, Self.orange)
                disc(w * 0.9, h + m * 0.38, m * 0.6, Self.pink)
            case 5:
                band(0, h, Self.ink)
                disc(w * 0.62, h * 0.38, m * 0.22, Self.blue)
                disc(w * 0.62 + m * 0.1, h * 0.38 - m * 0.06, m * 0.2, Self.ink)
                for (a, b) in [(0.2, 0.2), (0.3, 0.6), (0.15, 0.45), (0.85, 0.75), (0.42, 0.15), (0.8, 0.18)] as [(CGFloat, CGFloat)] {
                    disc(w * a, h * b, max(0.8, m * 0.012), Self.paper)
                }
                band(h * 0.84, h * 0.16, Self.teal)
            case 6:
                band(0, h, Self.blue)
                let r = m * 0.14
                for i in 0..<6 {
                    let angle = Double(i) / 6 * 2 * .pi
                    disc(w * 0.5 + CGFloat(cos(angle)) * r * 1.25, h * 0.5 + CGFloat(sin(angle)) * r * 1.25, r, Self.pink)
                }
                disc(w * 0.5, h * 0.5, r * 0.9, Self.yellow)
            default:
                band(0, h, Self.orange)
                let x = w * 0.56, y = h * 0.4, s = m * 0.24
                var kite = Path()
                kite.move(to: CGPoint(x: x, y: y - s))
                kite.addLine(to: CGPoint(x: x + s * 0.7, y: y))
                kite.addLine(to: CGPoint(x: x, y: y + s * 1.2))
                kite.addLine(to: CGPoint(x: x - s * 0.7, y: y))
                kite.closeSubpath()
                context.fill(kite, with: .color(Self.teal))
                var shade = Path()
                shade.move(to: CGPoint(x: x, y: y - s))
                shade.addLine(to: CGPoint(x: x, y: y + s * 1.2))
                shade.addLine(to: CGPoint(x: x + s * 0.7, y: y))
                shade.closeSubpath()
                context.fill(shade, with: .color(Self.ink.opacity(0.18)))
                var string = Path()
                string.move(to: CGPoint(x: x, y: y + s * 1.2))
                string.addCurve(to: CGPoint(x: w * 0.2, y: h * 1.02), control1: CGPoint(x: x - s, y: y + s * 2), control2: CGPoint(x: x + s * 0.4, y: h * 0.85))
                context.stroke(string, with: .color(Self.ink), lineWidth: max(1, m * 0.012))
            }
        }
        .accessibilityHidden(true)
    }
}

/// The SwiftPieces motion language: five spring tiers, and a named role for every moment a piece moves,
/// each with its Reduce Motion substitute.
///
/// Build one from the environment, `PieceMotion(reduceMotion: reduceMotion)`, and pick the role that names
/// what just happened. Never animate the tracking of a finger: set gesture state directly in `onChanged`, so the
/// surface stays under the finger, and spring only the release.
private struct PieceMotion {
    var reduceMotion = false

    // Tiers. Overshoot and settle times are measured from rest with SwiftUI's Spring.
    /// No overshoot, 90% in about 90ms. A press arriving under the finger.
    static var tight: Spring { Spring(duration: 0.14, bounce: 0) }
    /// About 2.8% overshoot, 90% in about 140ms. Snapping to a detent, page or segment.
    static var responsive: Spring { Spring(duration: 0.32, bounce: 0.25) }
    /// About 8.4% overshoot. Visible give: a release, a return from past an edge, a drag settling home.
    static var elastic: Spring { Spring(duration: 0.42, bounce: 0.38) }
    /// About 15% overshoot. A resolved action landing. At most once per interaction.
    static var expressive: Spring { Spring(duration: 0.48, bounce: 0.48) }
    /// About 1.5% overshoot, unhurried. Opening large surfaces and ambient change.
    static var calm: Spring { Spring(duration: 0.5, bounce: 0.2) }

    /// Reduce Motion: settles become this short spring with no overshoot.
    private static var still: Animation { .spring(duration: 0.25, bounce: 0) }

    // Roles.
    /// Touch-down. Starts on the same frame and never bounces under the finger.
    var press: Animation { .spring(Self.tight) }
    /// The finger lifts off a pressed surface, which springs back through rest.
    var release: Animation { reduceMotion ? Self.still : .spring(Self.elastic) }
    /// A dragged thing comes to rest. With a gesture's velocity, use `settle(velocity:from:to:)` instead.
    var settle: Animation { reduceMotion ? Self.still : .spring(Self.elastic) }
    /// Lands on a detent, page or segment.
    var snap: Animation { reduceMotion ? Self.still : .spring(Self.responsive) }
    /// A number, a chart value or anything else people read moves to its new value. Never overshoots, so it
    /// never shows a value that isn't true.
    var value: Animation { reduceMotion ? Self.still : .spring(duration: 0.35, bounce: 0) }
    /// Comes back from past a limit: a pull beyond the edge, a value pushed against its bound.
    var rebound: Animation { reduceMotion ? Self.still : .spring(Self.elastic) }
    /// A shape or container changes size, corner radius or form, with a little give at the end.
    var morph: Animation { reduceMotion ? Self.still : .spring(duration: 0.4, bounce: 0.2) }
    /// Something appears, opens or expands. Opening is a little slower than closing.
    var reveal: Animation { reduceMotion ? .easeOut(duration: 0.2) : .spring(Self.calm) }
    /// Something leaves, closes or collapses. Quick and firm, out of the way.
    var dismiss: Animation { reduceMotion ? .easeIn(duration: 0.18) : .spring(duration: 0.3, bounce: 0.08) }
    /// A resolved action lands: a check, a sent state, a reaction.
    var success: Animation { reduceMotion ? .easeOut(duration: 0.24) : .spring(Self.expressive) }
    /// A refused action. Firm, no wobble; `pieceShake` adds the movement.
    var error: Animation { reduceMotion ? .easeOut(duration: 0.2) : .spring(Self.responsive) }
    /// Slow ambient change. Loops themselves stop under Reduce Motion: check `allowsAmbient`.
    var ambient: Animation { reduceMotion ? .easeInOut(duration: 0.3) : .spring(Self.calm) }

    /// Loops, drifts, idle breathing and parallax run only when this is true.
    var allowsAmbient: Bool { !reduceMotion }

    /// A moving transition, or a plain fade under Reduce Motion.
    func transition(_ transition: AnyTransition) -> AnyTransition {
        reduceMotion ? .opacity : transition
    }

    /// For content replaced inside a container that stays put (a label, a glyph, a count): the old content
    /// blurs out as the new one sharpens in. A fade under Reduce Motion.
    @MainActor var swap: AnyTransition {
        reduceMotion ? .opacity : AnyTransition(.blurReplace)
    }
}

extension PieceMotion {
    /// Follow-through: rank 0 leads, each later rank arrives a beat later on a slightly looser spring. Safe to reverse.
    func follow(_ spring: Spring = PieceMotion.elastic, rank: Int) -> Animation {
        guard !reduceMotion else { return .spring(duration: 0.25, bounce: 0) }
        let k = Double(min(max(rank, 0), 6))
        return .spring(duration: spring.duration + 0.04 * k, bounce: min(spring.bounce + 0.02 * k, 0.55))
    }

    /// One-shot entrances only: item `index` waits 30ms per place, capped at the seventh. Exits go together.
    func cascade(_ animation: Animation, index: Int, step: Double = 0.03) -> Animation {
        guard !reduceMotion, index > 0 else { return animation }
        return animation.delay(step * Double(min(index, 7)))
    }
}

## Notes

- Wraps your own views: pass any `Identifiable` items and draw each photo in `content`. Each is clipped to `style.cornerRadius` and sized to a square, so images should fill (`.scaledToFill()`).
- The first four items make the pile, tilted 0°, -4°, 7° and -9° and nudged a few points apart; the rest wait hidden under it. A tap fans them all into a grid of `columns` columns.
- Each photo travels on its own spring a beat after the one before (`PieceMotion.follow(rank:)`), so the grid fills in reading order; gathering runs the other way round, so the last photo leaves first.
- It takes the width it is offered and grows to the height the grid needs, so put it in a `ScrollView` or a `VStack`. The count under the pile comes from `countLabel`; pass `nil` to hide it.
- Pass `onSelect` to open a photo tapped in the grid. Without it, a tap in the grid gathers the pile again.
- VoiceOver reads the pile as one button with its count, and the grid as separate photos, with Show all and Stack actions. A light haptic marks each toggle.
- Under Reduce Motion the photos move without bounce.
