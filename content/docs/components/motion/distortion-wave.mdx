import SwiftUI

/// Sizes and timing for `DistortionWave`.
public struct DistortionWaveStyle: Sendable {
    /// How long the ring takes to cross the screen, in seconds.
    public var duration: Double
    /// How wide the band of bending is around the ring, in points.
    public var band: CGFloat

    public init(duration: Double = 1.1, band: CGFloat = 40) {
        self.duration = duration
        self.band = band
    }

    public static let standard = DistortionWaveStyle()
}

/// Shows `content` until `isRevealed` turns on, then opens `next` through a ring of refraction from `origin`.
///
/// Requires `DistortionWave.metal` in the same target: its `distortionWaveOld` and `distortionWaveNew` kernels are
/// looked up through `ShaderLibrary`. Give the ring its starting point with `.distortionWaveOrigin($frame)` on
/// the button, or leave the frame empty to start from the centre.
///
/// - Parameters:
///   - isRevealed: Whether `next` shows.
///   - origin: The frame, in global coordinates, the ring opens from. Usually the button's.
///   - style: Sizes and timing.
///   - content: The screen being left.
///   - next: The screen arriving.
public struct DistortionWave<Content: View, Next: View>: View {
    public typealias Style = DistortionWaveStyle

    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    @Binding private var isRevealed: Bool
    @State private var progress: CGFloat = 0
    @State private var frame: CGRect = .zero
    @State private var waves = 0

    private let origin: CGRect
    private let style: Style
    private let content: Content
    private let next: Next

    public init(
        isRevealed: Binding<Bool>,
        from origin: CGRect = .zero,
        style: Style = .standard,
        @ViewBuilder content: () -> Content,
        @ViewBuilder next: () -> Next
    ) {
        self._isRevealed = isRevealed
        self.origin = origin
        self.style = style
        self.content = content()
        self.next = next()
        self._progress = State(initialValue: isRevealed.wrappedValue ? 1 : 0)
    }

    public var body: some View {
        let center = origin == .zero || frame == .zero
            ? CGPoint(x: frame.width / 2, y: frame.height / 2)
            : CGPoint(x: origin.midX - frame.minX, y: origin.midY - frame.minY)
        let reach = farthestCorner(from: center, in: frame.size) + style.band
        let moving = progress > 0 && progress < 1

        ZStack {
            next
                .modifier(DistortionWaveLayer(progress: progress, center: center, reach: reach, band: style.band, arriving: true, active: moving && !reduceMotion))
                .opacity(reduceMotion ? Double(progress) : (progress > 0 ? 1 : 0))
                .allowsHitTesting(isRevealed)
                .accessibilityHidden(!isRevealed)
            content
                .modifier(DistortionWaveLayer(progress: progress, center: center, reach: reach, band: style.band, arriving: false, active: moving && !reduceMotion))
                .opacity(reduceMotion ? Double(1 - progress) : (progress < 1 ? 1 : 0))
                .allowsHitTesting(!isRevealed)
                .accessibilityHidden(isRevealed)
        }
        .onGeometryChange(for: CGRect.self) { $0.frame(in: .global) } action: { frame = $0 }
        .onChange(of: isRevealed) { _, revealed in
            waves += 1
            withAnimation(reduceMotion ? .easeInOut(duration: 0.3) : .timingCurve(0.65, 0, 0.35, 1, duration: revealed ? style.duration : style.duration * 0.8)) {
                progress = revealed ? 1 : 0
            }
        }
        .sensoryFeedback(.impact(weight: .medium), trigger: waves)
    }

    private func farthestCorner(from point: CGPoint, in size: CGSize) -> CGFloat {
        let dx = max(point.x, size.width - point.x), dy = max(point.y, size.height - point.y)
        return hypot(dx, dy)
    }
}

/// One side of the ring: the old screen outside it or the new one inside, as a Metal layer effect.
private struct DistortionWaveLayer: ViewModifier, Animatable {
    var progress: CGFloat
    let center: CGPoint
    let reach: CGFloat
    let band: CGFloat
    let arriving: Bool
    let active: Bool

    nonisolated var animatableData: CGFloat {
        get { progress }
        set { progress = newValue }
    }

    func body(content: Content) -> some View {
        let radius = Float(progress * reach)
        let shader = arriving
            ? ShaderLibrary.distortionWaveNew(.float2(center), .float(radius), .float(Float(band)))
            : ShaderLibrary.distortionWaveOld(.float2(center), .float(radius), .float(Float(band)))
        content
            .layerEffect(shader, maxSampleOffset: CGSize(width: band, height: band), isEnabled: active)
    }
}

public extension View {
    /// Reports this view's global frame, so a `DistortionWave` can open from it.
    func distortionWaveOrigin(_ frame: Binding<CGRect>) -> some View {
        onGeometryChange(for: CGRect.self) { $0.frame(in: .global) } action: { frame.wrappedValue = $0 }
    }
}

#include <metal_stdlib>
#include <SwiftUI/SwiftUI_Metal.h>
using namespace metal;

// Distortion Wave: a ring of glass that opens from a point. Inside the ring the new screen is magnified toward
// the rim, as under the curved edge of a lens; just outside it the old screen is drawn in toward the ring. The
// colour channels bend by slightly different amounts, so the rim splits light a little, and a thin highlight runs
// along the edge. Helpers are file-local (static) so the kernels can share a target with other shaders.

namespace distortionWave {

/// The band narrows while the ring is small, so it never folds over its own centre.
static float bandFor(float radius, float band) {
    return band * min(1.0, radius / (2.0 * band)) + 1.0;
}

/// Samples the layer with each colour channel bent by a slightly different amount.
static half4 split(SwiftUI::Layer layer, float2 position, float2 direction, float shift) {
    float spread = shift * 0.12;
    half4 r = layer.sample(position - direction * (shift + spread));
    half4 g = layer.sample(position - direction * shift);
    half4 b = layer.sample(position - direction * (shift - spread));
    return half4(r.r, g.g, b.b, g.a);
}

/// Adds the rim's highlight (and a faint glow inside it) and the soft shade just outside it.
static half4 light(half4 color, float q, float s, bool inside) {
    float rim = 1.4;
    float highlight = exp(-(q * q) / (2.0 * rim * rim)) * 0.7 + (inside ? 0.10 * s * s * s : 0.0);
    float shade = inside ? 0.0 : 0.10 * s * s;
    half3 rgb = mix(color.rgb, half3(color.a), half(highlight));
    return half4(rgb * half(1.0 - shade), color.a);
}

}

// The screen being left: hidden inside the ring, drawn in toward it just outside.
[[ stitchable ]] half4 distortionWaveOld(float2 position, SwiftUI::Layer layer, float2 center, float radius, float band) {
    if (radius <= 0.0) { return layer.sample(position); }
    float2 offset = position - center;
    float d = max(length(offset), 1.0);
    float q = d - radius;
    if (q < 0.0) { return half4(0.0); }
    float width = distortionWave::bandFor(radius, band);
    if (q >= width) { return layer.sample(position); }
    float2 direction = offset / d;
    float s = 1.0 - q / width;
    half4 color = distortionWave::split(layer, position, direction, -width * 0.38 * s * s);
    return distortionWave::light(color, q, s, false);
}

// The screen arriving: shown inside the ring, magnified toward its rim.
[[ stitchable ]] half4 distortionWaveNew(float2 position, SwiftUI::Layer layer, float2 center, float radius, float band) {
    float2 offset = position - center;
    float d = max(length(offset), 1.0);
    float q = d - radius;
    if (q >= 0.0) { return half4(0.0); }
    float width = distortionWave::bandFor(radius, band);
    if (q <= -width) { return layer.sample(position); }
    float2 direction = offset / d;
    float s = 1.0 + q / width;
    half4 color = distortionWave::split(layer, position, direction, min(width * 0.55 * s * s, d * 0.8));
    return distortionWave::light(color, q, s, true);
}

Sizes and timing for `DistortionWave`.

## Notes

- Requires `DistortionWave.metal` in the same target. Its `distortionWaveOld` and `distortionWaveNew` kernels are looked up through `ShaderLibrary` and run as two layer effects, one on each screen.
- When `isRevealed` turns on, a ring opens from the button you mark with `.distortionWaveOrigin($frame)`: the new screen shows inside it, magnified toward the rim like the edge of a lens, the old one is drawn into it from outside, and a thin highlight with a touch of colour split runs along the edge.
- Set it back and the ring closes into the button. Without an origin it opens from the centre.
- The effects only run while the ring moves, so both screens are plain views at rest.
- Under Reduce Motion the screens crossfade instead.
