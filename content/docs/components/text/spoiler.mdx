import SwiftUI

/// Sizes for `.spoiler`.
public struct SpoilerStyle: Sendable {
    /// Specks per square point. More looks denser.
    public var density: Double
    /// The specks' size range, in points.
    public var speckSize: ClosedRange<CGFloat>
    /// How fast the specks drift, in points per second.
    public var drift: ClosedRange<CGFloat>
    /// How long the reveal takes, in seconds.
    public var revealDuration: Double

    public init(density: Double = 0.5, speckSize: ClosedRange<CGFloat> = 0.8...1.8, drift: ClosedRange<CGFloat> = 2.5...8.5, revealDuration: Double = 0.7) {
        self.density = density
        self.speckSize = speckSize
        self.drift = drift
        self.revealDuration = revealDuration
    }

    public static let standard = SpoilerStyle()
}

public extension View {
    /// Covers this view with drifting specks until it is tapped.
    ///
    /// The specks take the foreground style, so they match the text they cover. To hide part of a sentence, split
    /// it and put the hidden part in its own `Text`.
    ///
    /// - Parameters:
    ///   - isRevealed: Whether the content shows. A tap sets it; set it back to `false` to hide it again.
    ///   - style: Density, size and timing.
    func spoiler(isRevealed: Binding<Bool>, style: SpoilerStyle = .standard) -> some View {
        modifier(Spoiler(isRevealed: isRevealed, style: style))
    }

    /// Covers this view with drifting specks until it is tapped, and keeps track of that itself.
    func spoiler(style: SpoilerStyle = .standard) -> some View {
        modifier(SpoilerOwnState(style: style))
    }
}

/// The spoiler cover. Use it through `.spoiler`.
public struct Spoiler: ViewModifier {
    public typealias Style = SpoilerStyle

    @Environment(\.accessibilityReduceMotion) private var reduceMotion
    @Binding private var isRevealed: Bool
    @State private var origin: CGPoint = .zero
    @State private var radius: CGFloat = 0
    @State private var revealedAt: Date?
    @State private var coverOpacity: Double = 1
    @State private var settled = false
    @State private var reveals = 0

    private let style: Style

    init(isRevealed: Binding<Bool>, style: Style) {
        self._isRevealed = isRevealed
        self.style = style
    }

    public func body(content: Content) -> some View {
        content
            .mask {
                GeometryReader { proxy in
                    Circle()
                        .frame(width: radius * 2, height: radius * 2)
                        .position(origin)
                        .frame(width: proxy.size.width, height: proxy.size.height)
                }
            }
            .overlay {
                if !(isRevealed && settled) {
                    TimelineView(.animation(paused: reduceMotion && revealedAt == nil)) { timeline in
                        SpoilerSpecks(time: timeline.date.timeIntervalSinceReferenceDate, burst: revealedAt?.timeIntervalSinceReferenceDate, origin: origin, style: style)
                    }
                    .opacity(coverOpacity)
                    .allowsHitTesting(false)
                }
            }
            .contentShape(.rect)
            .onTapGesture(coordinateSpace: .local) { point in
                if !isRevealed { reveal(from: point) }
            }
            .onGeometryChange(for: CGSize.self) { $0.size } action: { size in
                if isRevealed { radius = hypot(size.width, size.height) }
                if revealedAt == nil { origin = CGPoint(x: size.width / 2, y: size.height / 2) }
            }
            .onChange(of: isRevealed) { _, shown in
                if !shown { hide() } else if revealedAt == nil { reveal(from: origin) }
            }
            .sensoryFeedback(.impact(weight: .light), trigger: reveals)
            .accessibilityElement(children: isRevealed ? .contain : .ignore)
            .accessibilityLabel(isRevealed ? "" : "Spoiler")
            .accessibilityHint(isRevealed ? "" : "Double tap to reveal")
            .accessibilityAddTraits(isRevealed ? [] : .isButton)
            .accessibilityAction { if !isRevealed { reveal(from: origin) } }
    }

    private func reveal(from point: CGPoint) {
        reveals += 1
        origin = point
        revealedAt = .now
        isRevealed = true
        radius = 0
        withAnimation(.easeOut(duration: reduceMotion ? 0.2 : style.revealDuration)) { radius = 2000 }
        settled = false
        Task { @MainActor in
            // Once the specks have flown, stop drawing them.
            try? await Task.sleep(for: .seconds(1.3))
            if isRevealed { settled = true }
        }
    }

    private func hide() {
        radius = 0
        revealedAt = nil
        settled = false
        coverOpacity = 0
        withAnimation(.easeInOut(duration: 0.5)) { coverOpacity = 1 }
    }
}

/// `.spoiler()` with its own revealed state.
private struct SpoilerOwnState: ViewModifier {
    let style: SpoilerStyle
    @State private var isRevealed = false

    func body(content: Content) -> some View {
        content.modifier(Spoiler(isRevealed: $isRevealed, style: style))
    }
}

/// The specks at one moment. Each speck lives a short life in a loop, born somewhere new each time, so nothing is
/// stored between frames. After a reveal they fly out from `origin` and fade.
private struct SpoilerSpecks: View {
    let time: Double
    let burst: Double?
    let origin: CGPoint
    let style: SpoilerStyle

    var body: some View {
        Canvas { context, size in
            let w = Double(size.width), h = Double(size.height)
            guard w > 1, h > 1 else { return }
            let count = min(Int(w * h * style.density), 3000)
            let edge = 3.5
            let lowSize = Double(style.speckSize.lowerBound), highSize = Double(style.speckSize.upperBound)
            let lowDrift = Double(style.drift.lowerBound), highDrift = Double(style.drift.upperBound)
            let now = burst ?? time
            let since = burst.map { max(0, time - $0) } ?? 0
            for i in 0..<count {
                let life = 0.9 + Self.random(i, 1) * 1.6
                let shift = Self.random(i, 2) * life
                let cycle = floor((now + shift) / life)
                let local = (now + shift) / life - cycle
                let seed = i &* 7919 &+ Int(cycle.truncatingRemainder(dividingBy: 100_000))
                let angle = Self.random(seed, 3) * 2 * .pi
                let speed = lowDrift + Self.random(seed, 4) * (highDrift - lowDrift)
                var x = Self.fold(Self.random(seed, 5) * w + cos(angle) * speed * local * life, w)
                var y = Self.fold(Self.random(seed, 6) * h + sin(angle) * speed * local * life, h)
                let brightness = 0.5 + Self.random(seed, 7) * 0.5
                var alpha = sin(.pi * local) * brightness
                alpha *= Self.smooth(0, edge, x) * Self.smooth(0, edge, w - x) * Self.smooth(0, edge * 0.8, y) * Self.smooth(0, edge * 0.8, h - y)
                if since > 0 {
                    // Blown away from the tap: fast near it, slower further off, slowing as they go.
                    let dx = x - Double(origin.x), dy = y - Double(origin.y)
                    let distance = max(hypot(dx, dy), 1)
                    let push = 40 + 260 * exp(-distance / 60) + Self.random(i, 8) * 60
                    let travel = push * (1 - exp(-0.9 * since)) / 0.9
                    x += dx / distance * travel + (Self.random(i, 9) - 0.5) * 30 * since
                    y += dy / distance * travel * 0.6 + (Self.random(i, 10) - 0.5) * 40 * since
                    alpha *= max(0, 1 - since / (0.5 + Self.random(i, 11) * 0.6))
                }
                guard alpha > 0.02 else { continue }
                let side = lowSize + Self.random(seed, 12) * (highSize - lowSize)
                context.opacity = alpha
                context.fill(Path(CGRect(x: x - side / 2, y: y - side / 2, width: side, height: side)), with: .foreground)
            }
        }
        .accessibilityHidden(true)
    }

    /// A repeatable random number in 0..<1 for an index and a channel.
    nonisolated private static func random(_ index: Int, _ channel: Int) -> Double {
        var z = UInt64(bitPattern: Int64(index &* 1_000_003 &+ channel &* 7_919))
        z = (z ^ (z >> 30)) &* 0xbf58476d1ce4e5b9
        z = (z ^ (z >> 27)) &* 0x94d049bb133111eb
        z ^= z >> 31
        return Double(z >> 11) / Double(1 << 53)
    }

    /// Bounces a position back inside 0...length, like a speck reflecting off the edges.
    nonisolated private static func fold(_ value: Double, _ length: Double) -> Double {
        let period = length * 2
        let m = value.truncatingRemainder(dividingBy: period)
        let wrapped = m < 0 ? m + period : m
        return wrapped > length ? period - wrapped : wrapped
    }

    nonisolated private static func smooth(_ a: Double, _ b: Double, _ x: Double) -> Double {
        let t = min(max((x - a) / (b - a), 0), 1)
        return t * t * (3 - 2 * t)
    }
}

// MARK: - Example

/// A film review with the ending hidden. Tap the specks to read it; Hide covers it again.
private struct SpoilerExample: View {
    @State private var revealed = false

    var body: some View {
        VStack(alignment: .leading, spacing: 14) {
            Text("Lake Ida").font(.title2.weight(.semibold))
            Text("A slow, warm film about a summer on the water, and the best thing I saw this year.")
                .font(.body.weight(.semibold))
                .foregroundStyle(.secondary)
            Text("The ending: the ferry never comes, and Maya rows the whole family home across the lake at dawn.")
                .font(.body.weight(.semibold))
                .spoiler(isRevealed: $revealed)
            Button(revealed ? "Hide again" : "Tap the specks to reveal") { revealed = false }
                .font(.subheadline.weight(.semibold))
                .foregroundStyle(Color(red: 1, green: 0, blue: 0))
                .disabled(!revealed)
        }
        .padding(24)
        .frame(width: 340)
        .frame(maxWidth: .infinity, maxHeight: .infinity)
        .background(Color(UIColor { @Sendable traits in
            traits.userInterfaceStyle == .dark ? UIColor(red: 0.071, green: 0.071, blue: 0.071, alpha: 1) : UIColor(red: 0.953, green: 0.949, blue: 0.933, alpha: 1)
        }))
    }
}

## Notes

- `.spoiler()` on any view covers it with fine drifting specks in the foreground style, so they match the text they cover. Use `.spoiler(isRevealed:)` to keep the state yourself and hide it again.
- Every speck is drawn from the time alone, in one `Canvas`: each lives a short life in a loop, born somewhere new each time, so nothing is stored between frames and the cover never repeats.
- Tap and the specks blow away from your finger while the content opens in a circle from the same spot. Hiding it again fades the shimmer back in.
- To hide part of a sentence, split it and put the hidden part in its own `Text`.
- VoiceOver reads it as a Spoiler button until it is revealed, then reads the content.
- Under Reduce Motion the specks hold still and the content opens quickly.
